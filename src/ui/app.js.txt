/* ==========================================================================
   NOTE ON THE FILE EXTENSION
   --------------------------------------------------------------------------
   This file is plain JavaScript, but it is stored as `.txt` on purpose.

   The Worker serves the whole panel as ONE inlined HTML document: it imports
   this file as raw text and concatenates it into the document's single inline
   script element. That keeps the panel at one request with zero external
   dependencies, which is the point (see docs/limits.md and docs/anti-ban.md).

   Because this text is inlined, it must never contain a literal opening or
   closing script tag — the closing form would truncate the page outright.
   tests/ui.test.js enforces that.

   esbuild resolves any `.js` import as an ES module *before* wrangler's Text
   module rule can apply, so a `.js` extension here fails the build with
   "No matching export ... for import default". The `.txt` suffix is
   unambiguous: wrangler treats it as Text by default, on every version, with no
   custom rules needed.

   Everything below is still ordinary JS — `node --check` it any time by
   copying it to a `.js` name.
   ========================================================================== */

  function h(tag, attrs) {
    var el = document.createElement(tag);
    if (attrs) {
      Object.keys(attrs).forEach(function (k) {
        var v = attrs[k];
        if (v === null || v === undefined || v === false) return;
        if (k === 'class') el.className = v;
        else if (k === 'html') el.innerHTML = v;
        else if (k === 'text') el.textContent = v;
        else if (k.slice(0, 2) === 'on' && typeof v === 'function') el.addEventListener(k.slice(2), v);
        else if (k === 'value') el.value = v;
        else if (k === 'checked') el.checked = !!v;
        else if (v === true) el.setAttribute(k, '');
        else el.setAttribute(k, v);
      });
    }
    for (var i = 2; i < arguments.length; i++) {
      var kid = arguments[i];
      if (kid === null || kid === undefined || kid === false) continue;
      if (Array.isArray(kid)) kid.forEach(function (x) { if (x) el.appendChild(typeof x === 'string' ? document.createTextNode(x) : x); });
      else el.appendChild(typeof kid === 'string' || typeof kid === 'number' ? document.createTextNode(String(kid)) : kid);
    }
    return el;
  }

  function icon(name) {
    var s = document.createElement('span');
    s.innerHTML = ICONS[name] || '';
    return s.firstChild || document.createElement('span');
  }

  function btn(label, opts) {
    opts = opts || {};
    var b = h('button', { class: 'btn ' + (opts.kind || ''), type: 'button', onclick: opts.onclick, disabled: opts.disabled });
    if (opts.icon) b.appendChild(icon(opts.icon));
    if (label) b.appendChild(document.createTextNode(label));
    return b;
  }

  function toast(msg, kind) {
    var box = document.getElementById('toasts');
    if (!box) return;
    var el = h('div', { class: 'toast ' + (kind || 'info'), text: msg });
    box.appendChild(el);
    setTimeout(function () {
      el.style.transition = 'opacity .3s, transform .3s';
      el.style.opacity = '0'; el.style.transform = 'translateY(8px)';
      setTimeout(function () { if (el.parentNode) el.remove(); }, 320);
    }, 2600);
  }

  function fmtBytes(n) {
    if (!n) return '0 B';
    var u = ['B', 'KB', 'MB', 'GB', 'TB']; var i = 0;
    while (n >= 1024 && i < u.length - 1) { n /= 1024; i++; }
    return (i === 0 ? n : n.toFixed(n < 10 ? 2 : 1)) + ' ' + u[i];
  }
  function fmtNum(n) { return (n || 0).toLocaleString('en-US'); }
  function esc(s) { return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) { return ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]; }); }

  async function api(path, opts) {
    opts = opts || {};
    var res = await fetch(path, {
      method: opts.method || 'GET',
      headers: opts.body ? { 'content-type': 'application/json' } : {},
      body: opts.body ? JSON.stringify(opts.body) : undefined,
      credentials: 'same-origin',
    });
    var data = null;
    try { data = await res.json(); } catch (e) { data = null; }
    if (res.status === 401 && state.authed) { state.authed = false; renderAuth(); throw new Error('unauthorized'); }
    if (!res.ok) {
      var m = (data && data.error) || ('HTTP ' + res.status);
      var err = new Error(m); err.status = res.status; err.data = data; throw err;
    }
    return data;
  }

  function modal(opts) {
    var back = h('div', { class: 'backdrop' });
    var box = h('div', { class: 'modal' + (opts.wide ? ' wide' : '') });

    if (opts.foot) box.appendChild(h('div', { class: 'modal-foot' }, opts.foot));
    back.appendChild(box);
    back.addEventListener('click', function (e) { if (e.target === back) close(); });
    document.body.appendChild(back);
    function close() { if (back.parentNode) back.remove(); document.removeEventListener('keydown', onKey); }
    function onKey(e) { if (e.key === 'Escape') close(); }
    document.addEventListener('keydown', onKey);
    var first = box.querySelector('input,select,textarea');
    if (first) setTimeout(function () { first.focus(); }, 40);
    return { close: close, box: box };
  }

  async function qrModal(text, title) {
    var box = h('div', { class: 'qrbox' }, h('span', { class: 'spin' }));
    var hint = h('div', { class: 'qr-hint' });
    var ref = {};
    ref.m = modal({
      title: title || t('qrTitle'),
      body: h('div', {}, box, hint,
        h('div', { class: 'mono-list', style: 'margin-top:12px;max-height:110px;overflow:auto', text: text })),
      foot: [btn(t('copy'), { icon: 'copy', onclick: function () { copyText(text); } }),
             btn(t('download'), { icon: 'download', onclick: function () {
               var svg = box.querySelector('svg');
               if (!svg) return toast(t('qrNothing'), 'err');
               download(new XMLSerializer().serializeToString(svg), 'luncher-qr.svg', 'image/svg+xml');
             } }),
             btn(t('close'), { kind: 'primary', onclick: function () { ref.m.close(); } })],
    });
    try {
      var res = await fetch('/api/qr?t=' + encodeURIComponent(text));
      if (!res.ok) {
        // Report what the server actually said. Guessing from the text length
        // produced a misleading message once the capacity limit was raised: a
        // 918-character PattNG link is perfectly encodable.
        var detail = '';
        try { detail = (await res.json()).error || ''; } catch (ignore) { /* non-JSON body */ }
        throw new Error(detail || ('HTTP ' + res.status));
      }
      var svgText = await res.text();
      box.innerHTML = svgText;

      // The server reports which error-correction level and version it settled
      // on. High versions are dense and genuinely hard for a phone camera, so
      // say so instead of presenting an unscannable square as a success.
      var version = Number(res.headers.get('x-luncher-qr-version') || 0);
      var ec = res.headers.get('x-luncher-qr-ec') || '';
      var size = res.headers.get('x-luncher-qr-size') || '';
      if (version >= 20) {
        hint.appendChild(notice('warn', t('qrDenseTitle'),
          esc(t('qrDenseText')).replace('{v}', String(version)).replace('{n}', String(size)).replace('{ec}', ec)));
      } else if (version) {
        hint.appendChild(h('div', { class: 'qr-meta',
          text: t('qrMeta') + ' · v' + version + ' · ' + size + '×' + size + ' · EC ' + ec }));
      }
    } catch (e) {
      box.innerHTML = '';
      box.style.background = 'transparent';
      box.appendChild(h('div', { style: 'color:#ffc9c9;font-size:13px;padding:16px;text-align:center', text: e.message }));
      hint.appendChild(notice('info', t('qrFailedTitle'), esc(t('qrFailedText'))));
    }
  }

    var ports = h('div', { class: 'card' }, h('div', { class: 'card-head' }, h('h3', { text: t('allowedPorts') })));
    ports.appendChild(notice('info', 'Hard platform limit',
      'Workers can only dial these TCP ports. Anything else is refused with a clear reason instead of an error 1101.'));
    var pc = h('div', { class: 'chips' });
    (o.workerPorts || []).forEach(function (p) { pc.appendChild(h('span', { class: 'chip' }, String(p))); });
    ports.appendChild(pc);
    el.appendChild(ports);

    var sanc = h('div', { class: 'card' });
    sanc.appendChild(h('div', { class: 'card-head' },
      h('h3', { text: t('sanctionedList') }),
      h('span', { class: 'sub' }, (o.sanctionedCount || 0) + ' domains')));
    sanc.appendChild(notice('info', 'Routing, not magic',
      'These destinations refuse you by <b>source IP</b>. Routing them through a relay is what unlocks them — fragmentation alone cannot.'));
    if (o.sanctionedDomains && o.sanctionedDomains.length) {
      var sc = h('div', { class: 'chips', style: 'margin-top:12px' });
      o.sanctionedDomains.forEach(function (d) { sc.appendChild(h('span', { class: 'chip', text: d })); });
      sanc.appendChild(sc);
    }
    el.appendChild(sanc);

    var err = h('div', { class: 'card' }, h('div', { class: 'card-head' }, h('h3', { text: 'About error 1101' })));
    err.appendChild(h('p', { style: 'color:var(--muted);font-size:13px;margin:0', html:
      '<code>1101</code> means <i>the Worker threw a JavaScript exception</i>. It is never Cloudflare blocking you and never a censorship signal — it is always a bug: an unguarded property, an unhandled promise rejection, a <code>send()</code> on a closed socket, or dialling a port Workers may not use. ' +
      'Every handler here is wrapped so those become ordinary HTTP responses. See <code>docs/error-1101.md</code>.' }));
    el.appendChild(err);
  };

  /* ------------------------------------------------------------ settings */

  VIEWS.settings = function (el) {
    var s = state.settings; if (!s) return;
    var card = h('div', { class: 'card' }, h('div', { class: 'card-head' }, h('h3', { text: t('settings') })));

    var name = h('input', { value: s.panelName || 'Luncher' });
    var title = h('input', { value: s.profileTitle || 'Luncher' });
    var ws = h('input', { value: s.wsPath || '/luncher', style: 'direction:ltr;font-family:var(--mono)' });
    var lsel = h('select');
    lsel.appendChild(h('option', { value: 'fa', text: 'فارسی', selected: s.language === 'fa' }));
    lsel.appendChild(h('option', { value: 'en', text: 'English', selected: s.language === 'en' }));
    var cache = h('input', { type: 'number', min: '0', max: '3600', value: s.subCacheSeconds });
    var maxl = h('input', { type: 'number', min: '1', max: '40', value: s.maxLinksPerSub });
    var ads = h('input', { type: 'checkbox', checked: !!s.blockAds });
    var sanc = h('input', { type: 'checkbox', checked: !!s.unlockSanctioned });
    var useHost = h('input', { type: 'checkbox', checked: s.useWorkerHostAsAddress !== false });
    var melli = h('input', { type: 'checkbox', checked: !!s.netMelli });
    var melliDns = h('input', { value: s.netMelliDns || '', style: 'direction:ltr;font-family:var(--mono)', placeholder: 'https://dns.bitdns.ir/dns-query' });
    var autodetect = h('input', { type: 'checkbox', checked: s.autoDetectClient !== false });

    card.appendChild(h('div', { class: 'field-row' }, field(t('panelName'), name), field(t('profileTitle'), title)));
    card.appendChild(field(t('wsPath'), ws,
      'The path clients dial. Changing it invalidates every existing config.'));
    card.appendChild(h('div', { class: 'field-row' },
      field(t('language'), lsel), field(t('subCache'), cache), field(t('maxLinks'), maxl)));

    card.appendChild(h('label', { class: 'switch' }, sanc, h('span', { class: 'track' }),
      h('span', { class: 'lbl' }, t('unlockSanctioned'),
        h('small', { text: 'Route Gemini / ChatGPT / Claude and similar through a relay when one is configured.' }))));
    card.appendChild(h('label', { class: 'switch' }, ads, h('span', { class: 'track' }),
      h('span', { class: 'lbl' }, t('blockAds'), h('small', { text: 'Adds a geosite ad-block rule to generated JSON configs.' }))));
    card.appendChild(h('label', { class: 'switch' }, useHost, h('span', { class: 'track' }),
      h('span', { class: 'lbl' }, 'Also offer the Worker hostname as an address',
        h('small', { text: 'Turn off to emit clean-IP configs only.' }))));

    card.appendChild(h('label', { class: 'switch' }, autodetect, h('span', { class: 'track' }),
      h('span', { class: 'lbl' }, t('autoDetect'),
        h('small', { text: t('autoDetectHint') }))));

    // Net melli gets its own heading because the honest explanation does not fit
    // in a one-line hint, and a toggle that overpromises is worse than no toggle.
    card.appendChild(h('div', { class: 'card-head' }, h('h3', { text: t('netMelliTitle') })));
    card.appendChild(h('label', { class: 'switch' }, melli, h('span', { class: 'track' }),
      h('span', { class: 'lbl' }, t('netMelli'),
        h('small', { text: t('netMelliHint') }))));
    card.appendChild(field(t('netMelliDns'), melliDns, t('netMelliDnsHint')));
    // notice() injects its text as HTML, so escape even though this string is a
    // static translation with no user input in it.
    card.appendChild(notice('warn', t('netMelliCaveatTitle'), esc(t('netMelliCaveat'))));

    var save = btn(t('save'), { kind: 'primary', icon: 'check' });
    save.addEventListener('click', async function () {
      save.disabled = true;
      try {
        var r = await api('/api/settings', { method: 'PUT', body: {
          panelName: name.value, profileTitle: title.value, wsPath: ws.value, language: lsel.value,
          subCacheSeconds: Number(cache.value), maxLinksPerSub: Number(maxl.value),
          blockAds: ads.checked, unlockSanctioned: sanc.checked, useWorkerHostAsAddress: useHost.checked,
          netMelli: melli.checked, netMelliDns: melliDns.value.trim(), autoDetectClient: autodetect.checked } });
        state.settings = r.settings; toast(t('saved'), 'ok'); renderShell();
      } catch (e) { toast(e.message, 'err'); } finally { save.disabled = false; }
    });
    card.appendChild(h('div', { style: 'margin-top:16px' }, save));
    el.appendChild(card);

    var bk = h('div', { class: 'card' }, h('div', { class: 'card-head' }, h('h3', { text: t('backup') })));
    var file = h('input', { type: 'file', accept: 'application/json', style: 'display:none' });
    file.addEventListener('change', async function () {
      var f = file.files && file.files[0]; if (!f) return;
      try {
        var r = await api('/api/backup', { method: 'POST', body: JSON.parse(await f.text()) });
        toast('Imported ' + r.imported + ' users', 'ok');
        await reload();
      } catch (e) { toast(e.message, 'err'); }
    });
    bk.appendChild(h('div', { style: 'display:flex;gap:9px;flex-wrap:wrap' },
      btn(t('exportBackup'), { icon: 'download', onclick: async function () {
        try {
          var r = await api('/api/backup');
          download('luncher-backup.json', JSON.stringify(r, null, 2), 'application/json');
          toast('Exported', 'ok');
        } catch (e) { toast(e.message, 'err'); }
      } }),
      btn(t('importBackup'), { icon: 'refresh', onclick: function () { file.click(); } }), file));
    el.appendChild(bk);

    var sec = h('div', { class: 'card' }, h('div', { class: 'card-head' }, h('h3', { text: t('changePass') })));
    var p1 = h('input', { type: 'password', placeholder: '••••••••', autocomplete: 'new-password' });
    var p2 = h('input', { type: 'password', placeholder: '••••••••', autocomplete: 'new-password' });
    var ch = btn(t('setPassword'), { kind: 'primary', icon: 'key' });
    ch.addEventListener('click', async function () {
      if ((p1.value || '').length < 8) { toast('At least 8 characters', 'err'); return; }
      if (p1.value !== p2.value) { toast('Passwords do not match', 'err'); return; }
      try {
        await api('/api/setup', { method: 'POST', body: { password: p1.value, change: true } });
        toast(t('saved'), 'ok'); p1.value = ''; p2.value = '';
      } catch (e) { toast(e.message, 'err'); }
    });
    sec.appendChild(h('div', { class: 'field-row' }, field('New password', p1), field('Repeat', p2)));
    sec.appendChild(ch);
    sec.appendChild(h('div', { class: 'hint', style: 'margin-top:9px', text:
      'If ADMIN_PASS_HASH is set in the Worker environment it always wins and cannot be changed here.' }));
    el.appendChild(sec);
  };

  /* ---------------------------------------------------------------- boot */

  async function boot() {
    try {
      var o = await api('/api/overview');
      state.overview = o; state.settings = o.settings;
      await loadUsers();
      state.authed = true;
      var hash = (location.hash || '').replace('#', '');
      if (hash && VIEWS[hash]) state.view = hash;
      document.title = ((o.settings && o.settings.panelName) || 'Luncher') + ' · Panel';
      renderShell();
    } catch (e) {
      if (e.message !== 'unauthorized') toast(e.message, 'err');
    }
  }

  window.addEventListener('hashchange', function () {
    var v = (location.hash || '').replace('#', '');
    if (v && VIEWS[v] && state.authed) go(v);
  });

  (async function init() {
    document.documentElement.lang = lang;
    document.documentElement.dir = lang === 'fa' ? 'rtl' : 'ltr';
    try {
      var st = await fetch('/api/state').then(function (r) { return r.ok ? r.json() : null; });
      if (st) {
        state.configured = !!st.configured;
        state.overview = { presets: st.presets, version: st.version };
        document.title = (st.panelName || 'Luncher') + ' · Panel';
      }
    } catch (e) { /* offline or 500 — fall through to the login screen */ }

    if (!state.configured) { renderAuth(); return; }
    try { await api('/api/me'); await boot(); }
    catch (e) { renderAuth(); }
  })();
})();
